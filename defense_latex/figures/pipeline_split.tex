% Pipeline-split hybrid: 2 micro-batches (A,B) on 2 CUDA streams, staggered by one
% stage so one stream's split SHUFFLE (comm) overlaps the other's CONV (compute).
% Each layer is shuffle -> conv; at any step exactly one all-to-all is in flight
% (single NCCL communicator). Drawn for L=3 layers.
% \input{figures/pipeline_split} inside a figure environment.
\begin{tikzpicture}[
    font=\small,
    blk/.style={draw, minimum height=0.6cm, minimum width=1.4cm, anchor=west, font=\scriptsize},
    comp/.style={blk, fill=blue!12},
    comm/.style={blk, fill=red!14},
    idle/.style={blk, draw=gray, dashed, fill=black!3},
    ax/.style={-{Latex[length=1.6mm]}, gray},
  ]
  \def\w{1.4}
  \def\ya{0}
  \def\yb{-0.8}

  \node[font=\scriptsize, anchor=east] at (-0.15,\ya) {micro-batch $A$};
  \node[font=\scriptsize, anchor=east] at (-0.15,\yb) {micro-batch $B$};

  % A: shuffle/conv per layer, starting at step 0
  \foreach \l in {0,1,2}{
    \node[comm] at ({(2*\l)*\w},\ya)   {shuffle $L_\l$};
    \node[comp] at ({(2*\l+1)*\w},\ya) {conv $L_\l$};
  }
  % B: same sequence, one step later
  \node[idle] at (0,\yb) {};
  \foreach \l in {0,1,2}{
    \node[comm] at ({(2*\l+1)*\w},\yb) {shuffle $L_\l$};
    \node[comp] at ({(2*\l+2)*\w},\yb) {conv $L_\l$};
  }

  % time axis with step ticks
  \draw[ax] (0,-1.4) -- ({7*\w+0.4},-1.4) node[right, font=\scriptsize, text=gray] {time};
  \foreach \t in {0,...,7}{ \draw[gray] ({\t*\w},-1.35) -- ({\t*\w},-1.45); }

  % overlapped steps (1..5): one stream computes while the other shuffles
  \draw[decorate, decoration={brace, amplitude=4pt, mirror}, green!40!black]
    ({1*\w},-1.6) -- ({6*\w},-1.6)
    node[midway, below=5pt, font=\scriptsize\itshape, text=green!40!black]
    {overlapped: in each step one stream computes while the other shuffles};

  % legend
  \node[comm, minimum width=0.35cm, minimum height=0.25cm, anchor=west] (lc) at ({7*\w+0.2},\ya) {};
  \node[font=\scriptsize, anchor=west] at (lc.east) {comm};
  \node[comp, minimum width=0.35cm, minimum height=0.25cm, anchor=west] (lp) at ({7*\w+0.2},\yb) {};
  \node[font=\scriptsize, anchor=west] at (lp.east) {compute};
\end{tikzpicture}
